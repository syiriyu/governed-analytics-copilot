# Phase 2: Python concepts

import = bring another library/module into my code

as = aliasing

from x import y = import one thing from another module

def = define a function

() = function arguments / calling functions

: = starts an indented block

{} = dictionary

[] = List, or selecting DataFrame columns

( )tuple = fixed group of values

for ... in ... = loop through values

'=' = assign a value

.append() = add item to a list

.map() = map values using a rule/dictionary

return = send the result out of a function

rng.integers() = generate random integers
